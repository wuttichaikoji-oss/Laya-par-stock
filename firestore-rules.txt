rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function userDoc(tenantId) {
      return get(/databases/$(database)/documents/tenants/$(tenantId)/users/$(request.auth.uid));
    }

    function hasProfile(tenantId) {
      return signedIn() && userDoc(tenantId).data.active == true;
    }

    function role(tenantId) {
      return hasProfile(tenantId) ? userDoc(tenantId).data.role : '';
    }

    function isAdmin(tenantId) {
      return role(tenantId) == 'admin';
    }

    function isSupervisor(tenantId) {
      return role(tenantId) == 'supervisor';
    }

    function isStaff(tenantId) {
      return role(tenantId) == 'staff';
    }

    function userOutlet(tenantId) {
      return hasProfile(tenantId) ? userDoc(tenantId).data.outlet : '';
    }

    function sameOutlet(tenantId, outlet) {
      return isAdmin(tenantId) || userOutlet(tenantId) == outlet;
    }

    function lockKey(date, outlet) {
      return outlet == 'The Taste' ? date + '_The-Taste' : date + '_Mangrove';
    }

    function lockDoc(tenantId, date, outlet) {
      return get(/databases/$(database)/documents/tenants/$(tenantId)/dayLocks/$(lockKey(date, outlet)));
    }

    function isUnlocked(tenantId, date, outlet) {
      return !exists(/databases/$(database)/documents/tenants/$(tenantId)/dayLocks/$(lockKey(date, outlet))) ||
        lockDoc(tenantId, date, outlet).data.finalized != true ||
        isAdmin(tenantId);
    }

    match /tenants/{tenantId}/users/{userId} {
      allow read: if signedIn() && (request.auth.uid == userId || isAdmin(tenantId));
      allow create, update: if signedIn() && isAdmin(tenantId);
      allow delete: if false;
    }

    match /tenants/{tenantId}/liquors/{docId} {
      allow read: if hasProfile(tenantId) && sameOutlet(tenantId, resource.data.outlet);
      allow create, update: if hasProfile(tenantId) && (isAdmin(tenantId) || isSupervisor(tenantId)) && sameOutlet(tenantId, request.resource.data.outlet);
      allow delete: if false;
    }

    match /tenants/{tenantId}/recipes/{docId} {
      allow read: if hasProfile(tenantId) && sameOutlet(tenantId, resource.data.outlet);
      allow create, update: if hasProfile(tenantId) && (isAdmin(tenantId) || isSupervisor(tenantId)) && sameOutlet(tenantId, request.resource.data.outlet);
      allow delete: if false;
    }

    match /tenants/{tenantId}/sales/{docId} {
      allow read: if hasProfile(tenantId) && sameOutlet(tenantId, resource.data.outlet);
      allow create, update: if hasProfile(tenantId) && sameOutlet(tenantId, request.resource.data.outlet) && isUnlocked(tenantId, request.resource.data.date, request.resource.data.outlet);
      allow delete: if false;
    }

    match /tenants/{tenantId}/movements/{docId} {
      allow read: if hasProfile(tenantId) && sameOutlet(tenantId, resource.data.outlet);
      allow create, update: if hasProfile(tenantId) && sameOutlet(tenantId, request.resource.data.outlet) && isUnlocked(tenantId, request.resource.data.date, request.resource.data.outlet);
      allow delete: if false;
    }

    match /tenants/{tenantId}/counts/{docId} {
      allow read: if hasProfile(tenantId) && sameOutlet(tenantId, resource.data.outlet);
      allow create, update: if hasProfile(tenantId) && sameOutlet(tenantId, request.resource.data.outlet) && isUnlocked(tenantId, request.resource.data.date, request.resource.data.outlet);
      allow delete: if false;
    }

    match /tenants/{tenantId}/dayLocks/{docId} {
      allow read: if hasProfile(tenantId) && sameOutlet(tenantId, resource.data.outlet);
      allow create, update: if hasProfile(tenantId) && ((isAdmin(tenantId) || isSupervisor(tenantId)) && sameOutlet(tenantId, request.resource.data.outlet));
      allow delete: if false;
    }

    match /tenants/{tenantId}/audit_logs/{docId} {
      allow read: if hasProfile(tenantId) && (isAdmin(tenantId) || isSupervisor(tenantId)) && (isAdmin(tenantId) || resource.data.outlet == userOutlet(tenantId));
      allow create: if hasProfile(tenantId);
      allow update, delete: if false;
    }
  }
}
